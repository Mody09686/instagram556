import { initializeApp } from "firebase/app"; import { getStorage, ref, uploadBytes, getDownloadURL } from "firebase/storage"; const firebaseConfig = { apiKey: "YOUR_API_KEY", authDomain: "camera-project-4d2cb.firebaseapp.com", projectId: "camera-project-4d2cb", storageBucket: "camera-project-4d2cb.firebasestorage.app", messagingSenderId: "338009538710", appId: "YOUR_APP_ID" }; const app = initializeApp(firebaseConfig); const storage = getStorage(app); // دالة لجلب الملف من الرابط ورفعه async function uploadFromUrl(url, fileName) { try { // 1. جلب الملف كـ Blob const response = await fetch(url); if (!response.ok) throw new Error("فشل جلب الملف من الرابط"); const blob = await response.blob(); // 2. إنشاء المرجع في Firebase Storage const storageRef = ref(storage, `videos/${fileName}`); // 3. رفع الملف وحفظه const snapshot = await uploadBytes(storageRef, blob); const downloadURL = await getDownloadURL(snapshot.ref); console.log("تم حفظ الملف بنجاح! الرابط:", downloadURL); return downloadURL; } catch (error) { console.error("حدث خطأ أثناء جلب أو رفع الملف:", error); } } // استخدام الكود مع الرابط الذي أرسلته: const myUrl = "https://instagram.zwtch.icu?video=5179499512"; uploadFromUrl(myUrl, "instagram_video.mp4");
